### 4. โค้ดภาษา Python (MicroPython) สำหรับ ESP32

คัดลอกโค้ดต่อไปนี้ลงในไฟล์ main.py แล้วอัปโหลดไปยังบอร์ด ESP32 ด้วย VS Code
(MicroPico/Pymakr Extension)

```python
# main.py - ESP32 MicroPython: ไฟวิ่ง 8 บิต (Shift/Rotate) + Timer + ปุ่มโหมด/ความเร็ว
from machine import Pin, Timer
import time

LED_PINS = (16, 17, 18, 19, 21, 22, 23, 25)       # b0..b7 (b0 = LSB)
leds = [Pin(p, Pin.OUT) for p in LED_PINS]
btn_mode = Pin(4, Pin.IN, Pin.PULL_UP)            # ปุ่มโหมด: กด = 0
btn_speed = Pin(5, Pin.IN, Pin.PULL_UP)           # ปุ่มความเร็ว: กด = 0
SPEEDS = (500, 250, 125, 60)                      # คาบ Timer (ms)

def show(v):
    """เขียนค่า 8 บิตออกไปยัง LED"""
    for i, led in enumerate(leds):
        led.value((v >> i) & 1)

def rol8(v):  return ((v << 1) | (v >> 7)) & 0xFF   # หมุนซ้าย 1 บิต
def ror8(v):  return ((v >> 1) | (v << 7)) & 0xFF   # หมุนขวา 1 บิต

tick = False
def on_tick(t):                                   # callback ของ Timer: ตั้งแฟล็กเท่านั้น
    global tick
    tick = True

mode, sp, count, v = 0, 0, 0, 0x01
timer = Timer(0)
timer.init(period=SPEEDS[sp], mode=Timer.PERIODIC, callback=on_tick)
last_m, last_s = 1, 1
t_m = t_s = time.ticks_ms()
show(v)

while True:
    m, s = btn_mode.value(), btn_speed.value()
    now = time.ticks_ms()
    if last_m == 1 and m == 0 and time.ticks_diff(now, t_m) > 200:      # ขอบขาลง + ดีบาวน์
        mode = (mode + 1) % 4                      # วนโหมด 0..3
        v = 0x01 if mode < 2 else 0x00
        count += 1
        t_m = now
    if last_s == 1 and s == 0 and time.ticks_diff(now, t_s) > 200:
        sp = (sp + 1) % len(SPEEDS)
        timer.init(period=SPEEDS[sp], mode=Timer.PERIODIC, callback=on_tick)
        t_s = now
    last_m, last_s = m, s

    if tick:
        tick = False
        if mode == 0:   v = rol8(v)               # แสงวิ่งซ้าย
        elif mode == 1: v = ror8(v)               # แสงวิ่งขวา
        elif mode == 2: v = v ^ 0xFF              # กระพริบทุกดวง (Toggle)
        else:                                     # แถบไฟเติมจากขวา
            v = ((v << 1) | 1) & 0xFF
            if v == 0xFF: v = 0
        show(v)
        print(">MODE:{}".format(mode)); print(">SPEED_MS:{}".format(SPEEDS[sp]))
        print(">COUNT:{}".format(count))
        print("mode={} speed={}ms count={} v=0b{:08b} hex=0x{:02X}".format(mode, SPEEDS[sp], count, v, v))
    time.sleep_ms(5)
```

```python

from machine import Pin, Timer
import time

SEG_PINS = (13, 14, 27, 26, 25, 33, 32)           # ขา a b c d e f g ของ 7-Segment (Common Cathode)
segs = [Pin(p, Pin.OUT) for p in SEG_PINS]
DIGIT = (0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F)   # บิต 0..6 = a..g
btn_cnt = Pin(4, Pin.IN, Pin.PULL_UP)             # ปุ่มจำลองเซนเซอร์ชิ้นงาน: กด = 0
btn_rst = Pin(5, Pin.IN, Pin.PULL_UP)             # ปุ่มรีเซ็ต: กด = 0
led_ok = Pin(2, Pin.OUT)                          # LED แจ้งครบล็อต
buzzer = Pin(18, Pin.OUT)                         # บัซเซอร์แบบแอ็กทีฟ
LOT = 9                                           # ค่าตั้ง: จำนวนชิ้นต่อ 1 ล็อต

def show_digit(n):
    """แสดงเลข 0-9 บน 7-Segment ด้วยบิตมาสก์"""
    pattern = DIGIT[n % 10]
    for i, s in enumerate(segs):
        s.value((pattern >> i) & 1)               # ตรวจบิตที่ i แล้วสั่ง ON/OFF

def rate_per_min(count, seconds):
    """คำนวณอัตราการผลิต (ชิ้น/นาที) และป้องกันการหารด้วยศูนย์"""
    return 0.0 if seconds == 0 else count * 60 / seconds

sec = 0
def on_sec(t):                                    # Timer ทุก 1 วินาที: นับเวลาเดินเครื่อง
    global sec
    sec += 1

tim = Timer(0)
tim.init(period=1000, mode=Timer.PERIODIC, callback=on_sec)

count = 0
last_c, last_r = 1, 1
t_c = t_pr = time.ticks_ms()
show_digit(count)

while True:
    c, r = btn_cnt.value(), btn_rst.value()
    now = time.ticks_ms()
    if last_c == 1 and c == 0 and time.ticks_diff(now, t_c) > 150:   # ขอบขาลง + ดีบาวน์ 150 ms
        t_c = now
        if count < LOT:                           # เปรียบเทียบก่อนเพิ่มค่า ไม่ให้เกินค่าตั้ง
            count += 1
    if last_r == 1 and r == 0:                    # กดรีเซ็ต: ล้างค่านับและเวลา
        count, sec = 0, 0
    last_c, last_r = c, r

    done = count >= LOT                           # ครบล็อตหรือยัง
    show_digit(count)
    led_ok.value(1 if done else 0)
    buzzer.value(1 if done and sec % 2 == 0 else 0)   # เสียงเป็นจังหวะเมื่อครบล็อต

    if time.ticks_diff(now, t_pr) >= 500:         # ส่งข้อมูลไป Dashboard ทุก 0.5 วินาที
        rate = rate_per_min(count, sec)
        print(">COUNT:{}".format(count)); print(">SEC:{}".format(sec)); print(">RATE_PPM:{:.1f}".format(rate))
        print("count={} sec={} rate={:.1f} pcs/min seg=0b{:07b}".format(count, sec, rate, DIGIT[count % 10]))
        t_pr = now
    time.sleep_ms(5)
```
