### 4. โค้ดภาษา Python (MicroPython) สำหรับ ESP32

คัดลอกโค้ดต่อไปนี้ลงในไฟล์ main.py แล้วอัปโหลดไปยังบอร์ด ESP32 ด้วย VS Code

```python
# main.py - ESP32 MicroPython: Self-diagnostic + log + Dashboard
from machine import Pin, I2C, ADC
import time

i2c = I2C(0, scl=Pin(22), sda=Pin(21), freq=100000)
pot = ADC(Pin(34)); pot.atten(ADC.ATTN_11DB)
btn = Pin(4, Pin.IN, Pin.PULL_UP)
led = Pin(2, Pin.OUT)
buz = Pin(27, Pin.OUT)

def log(msg):
    line = "{} {}".format(time.ticks_ms(), msg)
    print(line)
    try:
        with open('log.txt', 'a') as f:
            f.write(line + '\n')
    except OSError:
        print("LOG ERROR")

def check(name, ok, info=''):
    log("{} {} {}".format(name, 'PASS' if ok else 'FAIL', info))
    return 1 if ok else 0

def self_test():
    score = 0
    score += check('I2C_SHT30', 0x44 in i2c.scan())
    raw = pot.read()
    score += check('ADC_RANGE', 0 <= raw <= 4095, 'raw=%d' % raw)
    led.value(1); time.sleep_ms(200); led.value(0)
    score += check('LED', True, 'visual check')
    buz.value(1); time.sleep_ms(100); buz.value(0)
    score += check('BUZZER', True, 'audible check')
    return score, raw

log("BOOT")
while True:
    score, raw = self_test()
    print(">SCORE:{}".format(score))
    print(">ADC:{}".format(raw))
    print(">BUTTON:{}".format(1 - btn.value()))
    if btn.value() == 0:
        log("BUTTON pressed -> safe state")
        led.value(0); buz.value(0)
    time.sleep_ms(1000)
```

### 4. โค้ดภาษา Python (MicroPython) สำหรับ ESP32

คัดลอกโค้ดต่อไปนี้ลงในไฟล์ main.py แล้วอัปโหลดไปยังบอร์ด ESP32 ด้วย VS Code (MicroPico/Pymakr Extension)

```python
# main.py - ESP32 MicroPython: Watchdog Timer + สาเหตุการรีเซ็ต + ตัวนับการบูต + หน่วยความจำคงเหลือ
from machine import Pin, WDT, reset_cause
import machine, gc, json, time

btn = Pin(4, Pin.IN, Pin.PULL_UP)                    # กดค้างระหว่างทำงาน = จำลองโปรแกรมค้าง
led = Pin(2, Pin.OUT)
CAUSES = {machine.PWRON_RESET: "POWER_ON", machine.HARD_RESET: "HARD", machine.WDT_RESET: "WATCHDOG",
          machine.DEEPSLEEP_RESET: "DEEPSLEEP", machine.SOFT_RESET: "SOFT"}

def load_stats():
    try:
        with open("stats.json") as f:
            return json.load(f)
    except (OSError, ValueError):
        return {"boots": 0, "wdt": 0}

def save_stats(s):
    with open("stats.json", "w") as f:
        json.dump(s, f)

if btn.value() == 0:                                 # กดค้างขณะบูต = โหมดซ่อมบำรุง (ไม่เปิด WDT)
    print("MAINTENANCE MODE: WDT not started, REPL ready")
    raise SystemExit

st = load_stats()
cause = CAUSES.get(reset_cause(), "UNKNOWN")
st["boots"] += 1
if cause == "WATCHDOG":
    st["wdt"] += 1
save_stats(st)
print("BOOT #{} cause={} wdt_resets={}".format(st["boots"], cause, st["wdt"]))

wdt = WDT(timeout=3000)                              # ต้อง feed ภายใน 3 วินาที มิฉะนั้นบอร์ดรีเซ็ตเอง
loops = 0
t_pr = time.ticks_ms()
while True:
    if btn.value() == 0:                             # จำลองบั๊ก: วนค้างโดยไม่ feed WDT
        print("SIMULATED HANG ... waiting for watchdog")
        while True:
            led.value(1)
    wdt.feed()
    loops += 1
    led.value((loops // 25) % 2)                     # กะพริบแสดงว่าโปรแกรมยังทำงาน
    if time.ticks_diff(time.ticks_ms(), t_pr) >= 1000:
        gc.collect()
        print(">FREE_KB:{:.1f}".format(gc.mem_free() / 1024)); print(">LOOPS_S:{}".format(loops))
        print(">BOOTS:{}".format(st["boots"])); print(">WDT_RESETS:{}".format(st["wdt"]))
        loops = 0
        t_pr = time.ticks_ms()
    time.sleep_ms(10)
```
