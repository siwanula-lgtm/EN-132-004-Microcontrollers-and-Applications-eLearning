### 4. โค้ดภาษา Python (MicroPython) สำหรับ ESP32

คัดลอกไฟล์ ssd1306.py จาก MicroPython-lib ไปยังบอร์ดก่อน
แล้วคัดลอกโค้ดต่อไปนี้ลงในไฟล์ main.py

```python
# main.py - ESP32 MicroPython: DS18B20 + OLED SSD1306 + Keypad 4x4
from machine import Pin, I2C
import onewire, ds18x20, ssd1306, time

ds = ds18x20.DS18X20(onewire.OneWire(Pin(4)))
roms = ds.scan()
i2c = I2C(0, scl=Pin(22), sda=Pin(21))
oled = ssd1306.SSD1306_I2C(128, 64, i2c)

KEYS = [['1','2','3','A'], ['4','5','6','B'], ['7','8','9','C'], ['*','0','#','D']]
rows = [Pin(p, Pin.OUT, value=1) for p in (13, 12, 14, 27)]
cols = [Pin(p, Pin.IN, Pin.PULL_UP) for p in (26, 25, 33, 32)]
PIN_CODE = "1234"

def scan_key():
    for r in range(4):
        rows[r].value(0)
        for c in range(4):
            if cols[c].value() == 0:
                rows[r].value(1)
                return KEYS[r][c]
        rows[r].value(1)
    return None

def draw(temp, entered, msg):
    oled.fill(0)
    oled.text("T={:.2f} C".format(temp), 0, 0)
    oled.text("PIN:" + "*" * len(entered), 0, 20)
    oled.text(msg, 0, 40)
    oled.show()

buf, msg, last, t_last = "", "ENTER PIN", None, 0
temp = 0.0
if roms: ds.convert_temp()
t_conv = time.ticks_ms()
while True:
    if roms and time.ticks_diff(time.ticks_ms(), t_conv) >= 750:
        temp = ds.read_temp(roms[0]); ds.convert_temp()
        t_conv = time.ticks_ms()
        print(">TEMP_C:{:.2f}".format(temp))
    k = scan_key()
    if k and k != last and time.ticks_diff(time.ticks_ms(), t_last) > 30:
        t_last = time.ticks_ms()
        if k == '#':
            msg = "ACCESS OK" if buf == PIN_CODE else "WRONG PIN"
            print(">PIN_OK:{}".format(1 if buf == PIN_CODE else 0)); buf = ""
        elif k == '*':
            buf, msg = "", "ENTER PIN"
        elif len(buf) < 8:
            buf += k
        print(">KEYS:{}".format(len(buf)))
    last = k
    draw(temp, buf, msg)
    time.sleep_ms(20)
```

### 4. โค้ดภาษา Python (MicroPython) สำหรับ ESP32

คัดลอกโค้ดต่อไปนี้ลงในไฟล์ main.py แล้วอัปโหลดไปยังบอร์ด ESP32 ด้วย VS Code (MicroPico/Pymakr Extension)

```python
# main.py - ESP32 MicroPython: HC-SR04 (ระยะทาง) + BH1750 (แสง, I2C) + OLED + ตรวจการเดินสาย
from machine import Pin, I2C, time_pulse_us
import ssd1306, time

trig = Pin(5, Pin.OUT, value=0)
echo = Pin(18, Pin.IN)                               # ผ่านวงจรแบ่งแรงดัน 1 kΩ/2 kΩ (5 V → 3.3 V)
i2c = I2C(0, scl=Pin(22), sda=Pin(21), freq=100000)
BH1750, OLED = 0x23, 0x3C

def check_bus():
    """ตรวจอุปกรณ์บนบัส I2C แล้วรายงานตัวที่หายไป"""
    found = i2c.scan()
    for name, addr in (("BH1750", BH1750), ("OLED", OLED)):
        print("{} 0x{:02X} {}".format(name, addr, "OK" if addr in found else "MISSING"))
    return found

def distance_cm():
    """ส่งพัลส์ Trigger 10 µs แล้ววัดความกว้างพัลส์ Echo"""
    trig.value(0); time.sleep_us(2)
    trig.value(1); time.sleep_us(10); trig.value(0)
    t = time_pulse_us(echo, 1, 30000)                # หมดเวลา 30 ms (ประมาณ 5 m)
    return None if t < 0 else t * 0.0343 / 2         # ความเร็วเสียง 343 m/s คิดระยะไป-กลับ

def lux():
    """อ่าน BH1750 โหมด Continuous High-Resolution (ค่าดิบ / 1.2 = lx)"""
    d = i2c.readfrom(BH1750, 2)
    return ((d[0] << 8) | d[1]) / 1.2

found = check_bus()
if BH1750 in found:
    i2c.writeto(BH1750, b'\x01'); i2c.writeto(BH1750, b'\x10')   # Power on + วัดต่อเนื่อง
oled = ssd1306.SSD1306_I2C(128, 64, i2c) if OLED in found else None
buf = []
t_pr = time.ticks_ms()
while True:
    d = distance_cm()
    if d is not None:
        buf.append(d); buf = buf[-5:]                # ค่าเฉลี่ยเคลื่อนที่ 5 ค่า
    d_avg = sum(buf) / len(buf) if buf else 0
    L = lux() if BH1750 in found else 0
    if oled:
        oled.fill(0)
        oled.text("DIST {:6.1f} cm".format(d_avg), 0, 0)
        oled.text("LIGHT {:6.0f} lx".format(L), 0, 16)
        oled.rect(0, 36, 128, 12, 1)
        oled.fill_rect(0, 36, int(min(d_avg, 100) * 1.28), 12, 1)   # แถบระยะ 0-100 cm
        oled.text("ECHO ERR" if d is None else "ECHO OK", 0, 54)
        oled.show()
    if time.ticks_diff(time.ticks_ms(), t_pr) >= 250:
        print(">DIST_CM:{:.1f}".format(d_avg)); print(">LUX:{:.0f}".format(L))
        print(">ECHO_OK:{}".format(0 if d is None else 1))
        t_pr = time.ticks_ms()
    time.sleep_ms(60)                                # HC-SR04 ควรเว้นช่วงการวัดอย่างน้อย 60 ms
```
